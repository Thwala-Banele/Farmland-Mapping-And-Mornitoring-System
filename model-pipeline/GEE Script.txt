 1. Core Geographic Boundaries
var eswatini = ee.FeatureCollection(USDOSLSIB_SIMPLE2017)
  .filter(ee.Filter.or(ee.Filter.eq('country_na', 'Eswatini'), ee.Filter.eq('country_na', 'Swaziland')));

var newFieldBoundaries = ee.FeatureCollection('projectscalcium-post-439902-k6assetslubombo_fields_new');
var eswatiniAdminBoundaries = ee.FeatureCollection(FAOGAUL2015level2).filterBounds(eswatini);

 2. Pre-Calculate Area, Administrative Spatial Joins, and Native Coordinate Arrays
var masterLocalizedFields = newFieldBoundaries.map(function(field) {
  var intersectingAdmin = eswatiniAdminBoundaries.filterBounds(field.geometry()).first();
  var regionName = ee.Algorithms.If(intersectingAdmin, intersectingAdmin.get('ADM1_NAME'), 'Unknown_Region');
  var constituencyName = ee.Algorithms.If(intersectingAdmin, intersectingAdmin.get('ADM2_NAME'), 'Unknown_Inkhundla');
  
   Calculate field size dynamically in hectares
  var areaHectares = field.geometry().area().divide(10000); 
  
   FIXED Extract the raw nested array of coordinate points natively supported by LeafletMapbox
  var rawCoordinatesArray = field.geometry().coordinates();

  return field.set({
    'field_id' field.get('systemindex'),
    'region' regionName,
    'constituency' constituencyName,
    'area_ha' areaHectares,
    'geometry_coords' rawCoordinatesArray
  });
});

 Define strict time boundaries for Part 1
var startDate = '2017-07-01';
var endDate = '2021-12-31';

 3. High-Performance Multi-Satellite Index Engine
function maskClouds(image) {
  var qa = image.select('QA60');
  var mask = qa.bitwiseAnd(1  10).eq(0).and(qa.bitwiseAnd(1  11).eq(0));
  return image.updateMask(mask).divide(10000)
    .copyProperties(image, [systemtime_start]);
}

function calculateIndices(image) {
  var b2 = image.select('B2');  
  var b3 = image.select('B3');  
  var b4 = image.select('B4');  
  var b8 = image.select('B8');  
  var b11 = image.select('B11'); 

  var ndvi = b8.subtract(b4).divide(b8.add(b4)).rename('ndvi');
  var gndvi = b8.subtract(b3).divide(b8.add(b3)).rename('gndvi');
  var evi = b8.subtract(b4).multiply(2.5).divide(b8.add(b4.multiply(6)).subtract(b2.multiply(7.5)).add(1)).rename('evi');
  var evi2 = b8.subtract(b4).multiply(2.5).divide(b8.add(b4.multiply(2.4)).add(1)).rename('evi2');
  var ndwi = b8.subtract(b11).divide(b8.add(b11)).rename('ndwi');
  var lswi = b8.subtract(b11).divide(b8.add(b11)).rename('lswi'); 

  return image.addBands([ndvi, gndvi, evi, evi2, ndwi, lswi]);
}

 Build core imagery asset query catalog
var sentinelCollection = ee.ImageCollection('COPERNICUSS2_SR_HARMONIZED')
  .filterDate(startDate, endDate)
  .filterBounds(masterLocalizedFields)
  .map(maskClouds)
  .map(calculateIndices);

 Pre-filter the MODIS LST collection to save memory cache
var modisLSTCollection = ee.ImageCollection('MODIS061MOD11A1')
  .filterDate(startDate, endDate)
  .select('LST_Day_1km');

 4. Lightweight Sequential Chronological Loop (2017 to 2021 Only)
var startYear = 2017;
var endYear = 2021;
var years = ee.List.sequence(startYear, endYear);
var months = ee.List.sequence(1, 12);

var monthlyClimateList = years.map(function(y) {
  return months.map(function(m) {
    var start = ee.Date.fromYMD(y, m, 1);
    var end = start.advance(1, 'month');
    
     Sentinel median calculation for vegetation metrics
    var s2Filtered = sentinelCollection.filterDate(start, end);
    var s2Count = s2Filtered.size();
    var s2Median = s2Filtered.median();
    
     Pull a 30-day mean window of MODIS LST to fill cloudy null gaps
    var lstFiltered = modisLSTCollection.filterDate(start, end);
    var lstMeanCelsius = lstFiltered.mean().multiply(0.02).subtract(273.15).rename('lst_celsius');
    
     Combine everything together into a clean month card
    var integratedMonthlyImage = s2Median.addBands(lstMeanCelsius);
    
    var dateString = ee.String(ee.Number(y).format('%d')).cat('_')
                      .cat(ee.String(ee.Number(m).format('%02d')));
    
    return integratedMonthlyImage.set({
      'systemtime_start' start.millis(),
      'date_label' dateString,
      'image_count' s2Count
    });
  });
}).flatten();

var monthlyClimateStack = ee.ImageCollection(monthlyClimateList)
  .filter(ee.Filter.gt('image_count', 0))
  .filter(ee.Filter.and(
    ee.Filter.gte('systemtime_start', ee.Date(startDate).millis()),
    ee.Filter.lte('systemtime_start', ee.Date(endDate).millis())
  ));

print('Master Monolith Part 1 Armed! Total Active Months', monthlyClimateStack.size());

 5. Streamlined Zonal Reduction Engine
var historicalFieldTable = monthlyClimateStack.map(function(monthlyImage) {
  var dateStr = monthlyImage.get('date_label');
  
  return monthlyImage.reduceRegions({
    collection masterLocalizedFields.select(['field_id', 'region', 'constituency', 'area_ha', 'geometry_coords']),
    reducer ee.Reducer.mean(), 
    scale 10,
    tileScale 16
  }).map(function(feature) {
    return feature.set({'observation_date' dateStr});
  });
}).flatten();

 6. Map to clean attributes for spreadsheet output
var cleanExportCollection = masterFieldTable.map(function(feature) {

    'ndvi' feature.get('ndvi'),
    'gndvi' feature.get('gndvi'),

    'evi2' feature.get('evi2'),
    'ndwi' feature.get('ndwi'),
    'lswi' feature.get('lswi'),
    'lst_celsius' feature.get('lst_celsius'),
    'geometry_coords' feature.get('geometry_coords')  Coordinate arrays map smoothly to strings in CSV
  });
});
